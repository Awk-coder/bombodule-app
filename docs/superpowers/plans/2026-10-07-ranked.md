# Ranked Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a Ranked tab: IIUM result-slip import, app-computed semester and degree tiers shown as MCSR-style pixel badges, a task-driven season Elo, and 46 achievements.

**Architecture:** Pure domain modules (`grades.ts`, `elo.ts`, `achievements.ts`) compute everything from `AppState`; nothing derived is stored except achievement unlock times. A new result-slip parser feeds `state.results`. UI is one new screen plus small edits to Tasks, TaskSheet, Home, TabBar and App.

**Tech Stack:** Preact 10, TypeScript, Vite, Vitest (+ happy-dom), Capacitor 8, pdfjs-dist, `@fontsource/pixelify-sans` (OFL).

**Spec:** `docs/superpowers/specs/2026-10-07-ranked-design.md`

## Global Constraints

- Tiers: S 3.67+ Netherite · A 3.50+ Diamond · B 3.00+ Emerald · C 2.50+ Gold · D 2.00+ Iron · F below 2.00 Coal.
- Elo divisions: Coal 0/400/500 · Iron 600/700/800 · Gold 900/1000/1100 · Emerald 1200/1300/1400 · Diamond 1500/1650/1800 · Netherite 2000+.
- GPA → Elo: GPA `0, 2.00, 2.50, 3.00, 3.50, 3.67, 4.00` → Elo `0, 600, 900, 1200, 1500, 2000, 2300`, linear between.
- Elo deltas: on time +15 (final +45), late −10, overdue not done −20 (final −60), floor 0. Unrated placement 900.
- Name, IC/passport and matric number from a result slip are never stored.
- Badge art is drawn for Hadir (pixel grids in code), never MCSR image files.
- Semester labels use the existing format `Sem 2, 2025/2026`.
- Tests run with `npm test` (TZ=Asia/Kuala_Lumpur). Match the surrounding code style: terse, few comments, named exports.
- Commit messages end with `Claude-Session: https://claude.ai/code/session_01H7YgbZnA3QatQEQipTa4M2`.

## Review Focus

1. Re-importing the same semester's slip — must replace, not duplicate, and flag `regraded` only when a grade differs (Task 4 test).
2. A slip with a non-letter grade (`W`, `*`, `EX`) — parsed, excluded from GPA, never a crash (Tasks 2–3 tests).
3. Unticking a task — `completedAt` cleared, Elo change disappears (Tasks 7–8 tests).
4. Old saved state without the new fields — loads with defaults (Task 1 test).
5. Picking the timetable PDF in the result importer (or vice versa) — clear error, nothing saved (Task 2 test).

```ts
import type { ResultCourse, SemesterResult } from "../domain/types";

export interface ParsedResult {
  label: string;
  courses: ResultCourse[];
  printed: SemesterResult["printed"];
  status: string;
  remarks: string;
}

const COURSE = /^([A-Z]{3,4} ?\d{4}[A-Z]?)\s+(.+?)\s+(\d+(?:\.\d+)?)\s+([A-Z]{1,2}[+-]?|\*)(?:\s+(\d*\.\d+))?(?:\s+(\d*\.\d+))?$/;
const FIELDS = ["TOTAL CREDIT POINTS", "CUMULATIVE GRADE POINTS", "TOTAL CREDIT HOURS", "CUMULATIVE CREDIT HOURS", "GRADE POINT AVERAGE(GPA)", "CGPA", "STATUS", "REMARKS"];

/** Read the i-Ma'luum "Examination Result" slip. Only labelled fields are read, so personal lines are never kept. */
export function parseResults(rows: string[][]): ParsedResult | null {
  let label: string | null = null;
  const courses: ResultCourse[] = [];
  const f: Record<string, string> = {};
  for (const cells of rows) {
    const text = cells.join(" ").replace(/\s+/g, " ").trim();
    const s = text.match(/SESSION\s*:\s*(\d{4}\/\d{4})\s+SEMESTER\s*:\s*(\d)/i);
    if (s) {
      label = `Sem ${s[2]}, ${s[1]}`;
      continue;
    }
    const c = text.match(COURSE);
    if (c) {
      courses.push({ code: c[1].replace(/^([A-Z]+)(\d)/, "$1 $2"), title: c[2], credits: Number(c[3]), grade: c[4], point: c[5] ? Number(c[5]) : null });
      continue;
    }
    cells.forEach((cell, i) => {
      const key = cell.trim().toUpperCase().replace("CUMMULATIVE", "CUMULATIVE");
      if (!FIELDS.includes(key)) return;
      const value = cells.slice(i + 1).find((x) => x.trim() !== ":");
      if (value) f[key] = value.replace(/^:\s*/, "").trim();
    });
  }
  const gpa = Number(f["GRADE POINT AVERAGE(GPA)"]);
  const cgpa = Number(f["CGPA"]);
  if (!label || courses.length === 0 || !f["CGPA"] || Number.isNaN(gpa) || Number.isNaN(cgpa)) return null;
  return {
    label,
    courses,
    printed: {
      gpa,
      cgpa,
      credits: Number(f["TOTAL CREDIT HOURS"] ?? 0),
      cumCredits: Number(f["CUMULATIVE CREDIT HOURS"] ?? 0),
      cumPoints: Number(f["CUMULATIVE GRADE POINTS"] ?? 0),
    },
    status: f["STATUS"] ?? "",
    remarks: f["REMARKS"] ?? "",
  };
}
```

```ts
import type { ResultCourse, SemesterResult } from "./types";

export type Tier = "S" | "A" | "B" | "C" | "D" | "F";
export type Ore = "netherite" | "diamond" | "emerald" | "gold" | "iron" | "coal";

export const TIERS: { tier: Tier; ore: Ore; minGpa: number }[] = [
  { tier: "S", ore: "netherite", minGpa: 3.67 },
  { tier: "A", ore: "diamond", minGpa: 3.5 },
  { tier: "B", ore: "emerald", minGpa: 3 },
  { tier: "C", ore: "gold", minGpa: 2.5 },
  { tier: "D", ore: "iron", minGpa: 2 },
  { tier: "F", ore: "coal", minGpa: 0 },
];

export const round2 = (x: number) => Math.round((x + Number.EPSILON) * 100) / 100;
export const tierOf = (g: number) => TIERS.find((t) => round2(g) >= t.minGpa)!;

export const LETTER_POINTS: Record<string, number> = { A: 4, "A-": 3.67, "B+": 3.33, B: 3, "B-": 2.67, "C+": 2.33, C: 2, D: 1.67, "D-": 1.33, E: 1, F: 0 };
const ZERO = new Set(["Y", "FR"]); // barred and fail-on-resit count as F
const UNCOUNTED = new Set(["P", "PA", "W", "AU", "EX", "IP", "I", "AP", "CO", "PR", "FA", "*"]);

/** Grade point that counts toward GPA, or null when this grade doesn't count. */
export function countedPoint(c: ResultCourse): number | null {
  if (ZERO.has(c.grade)) return 0;
  return LETTER_POINTS[c.grade] ?? null;
}

export const isFail = (c: ResultCourse) => {
  const p = countedPoint(c);
  return p !== null && p < 2;
};

export function gpa(courses: ResultCourse[]): { gpa: number; credits: number; points: number } {
  let credits = 0;
  let points = 0;
  for (const c of courses) {
    const p = countedPoint(c);
    if (p === null) continue;
    credits += c.credits;
    points += c.credits * p;
  }
  return { gpa: credits ? round2(points / credits) : 0, credits: round2(credits), points: round2(points) };
}

/** "Sem 2, 2025/2026" → "2025-2", which sorts chronologically. */
export function labelKey(label: string): string {
  const m = label.match(/Sem (\d), (\d{4})\/\d{4}/);
  return m ? `${m[2]}-${m[1]}` : label;
}

export const sortResults = (results: SemesterResult[]) => [...results].sort((a, b) => labelKey(a.label).localeCompare(labelKey(b.label)));

/** IIUM §15–16: a passed retake replaces a fail; retaking a passed course keeps the better grade. */
export function cgpa(results: SemesterResult[]): { cgpa: number; credits: number; points: number } {
  const attempts = new Map<string, ResultCourse[]>();
  for (const r of sortResults(results)) {
    for (const c of r.courses) {
      if (countedPoint(c) === null) continue;
      attempts.set(c.code, [...(attempts.get(c.code) ?? []), c]);
    }
  }
  const kept = [...attempts.values()].map((list) => {
    const passed = list.filter((c) => !isFail(c));
    return passed.length ? passed.reduce((a, b) => (countedPoint(b)! > countedPoint(a)! ? b : a)) : list[list.length - 1];
  });
  const g = gpa(kept);
  return { cgpa: g.gpa, credits: g.credits, points: g.points };
}

export function unknownGrades(results: SemesterResult[]): string[] {
  const out = new Set<string>();
  for (const r of results) for (const c of r.courses) if (countedPoint(c) === null && !UNCOUNTED.has(c.grade)) out.add(c.grade);
  return [...out];
}

export interface Degree {
  cgpa: number;
  credits: number;
  ok: boolean;
  reason: string | null;
}

/** The app's own CGPA; `ok` only when it reproduces the latest slip, i.e. every past slip is imported. */
export function degree(results: SemesterResult[]): Degree | null {
  if (!results.length) return null;
  const latest = sortResults(results).at(-1)!;
  const c = cgpa(results);
  const unknown = unknownGrades(results);
  let reason: string | null = null;
  if (unknown.length) reason = `Unknown grade codes: ${unknown.join(", ")}`;
  else if (c.credits !== latest.printed.cumCredits) reason = `Found ${c.credits} of ${latest.printed.cumCredits} credit hours. Import your earlier result slips.`;
  else if (c.cgpa !== latest.printed.cgpa) reason = `Calculated CGPA ${c.cgpa.toFixed(2)}, slip says ${latest.printed.cgpa.toFixed(2)}.`;
  return { cgpa: c.cgpa, credits: c.credits, ok: !reason, reason };
}

export function semesterMismatch(r: SemesterResult): string | null {
  const g = gpa(r.courses).gpa;
  return g === r.printed.gpa ? null : `Calculated GPA ${g.toFixed(2)}, slip says ${r.printed.gpa.toFixed(2)}.`;
}

const GPA_PTS = [0, 2, 2.5, 3, 3.5, 3.67, 4];
const ELO_PTS = [0, 600, 900, 1200, 1500, 2000, 2300];

export function eloFromGpa(g: number): number {
  const x = Math.min(4, Math.max(0, round2(g)));
  for (let i = 1; i < GPA_PTS.length; i++) {
    if (x <= GPA_PTS[i]) return Math.round(ELO_PTS[i - 1] + ((x - GPA_PTS[i - 1]) / (GPA_PTS[i] - GPA_PTS[i - 1])) * (ELO_PTS[i] - ELO_PTS[i - 1]));
  }
  return ELO_PTS[ELO_PTS.length - 1];
}

export const DIVISIONS: { ore: Ore; div: 1 | 2 | 3 | null; min: number }[] = [
  { ore: "coal", div: 1, min: 0 }, { ore: "coal", div: 2, min: 400 }, { ore: "coal", div: 3, min: 500 },
  { ore: "iron", div: 1, min: 600 }, { ore: "iron", div: 2, min: 700 }, { ore: "iron", div: 3, min: 800 },
  { ore: "gold", div: 1, min: 900 }, { ore: "gold", div: 2, min: 1000 }, { ore: "gold", div: 3, min: 1100 },
  { ore: "emerald", div: 1, min: 1200 }, { ore: "emerald", div: 2, min: 1300 }, { ore: "emerald", div: 3, min: 1400 },
  { ore: "diamond", div: 1, min: 1500 }, { ore: "diamond", div: 2, min: 1650 }, { ore: "diamond", div: 3, min: 1800 },
  { ore: "netherite", div: null, min: 2000 },
];

export interface Rank {
  ore: Ore;
  div: 1 | 2 | 3 | null;
  min: number;
  next: number | null;
  progress: number; // 0..1 toward `next`
}

export function rankOf(elo: number): Rank {
  let i = 0;
  while (i + 1 < DIVISIONS.length && elo >= DIVISIONS[i + 1].min) i++;
  const d = DIVISIONS[i];
  const next = DIVISIONS[i + 1]?.min ?? null;
  return { ...d, next, progress: next === null ? 1 : (elo - d.min) / (next - d.min) };
}
```

```ts
/** Save a semester's result slip; re-importing replaces it and notes whether any grade changed. */
export function importResult(state: AppState, parsed: ParsedResult, now: Date): AppState {
  const prev = state.results.find((r) => r.label === parsed.label);
  const grade = (code: string) => prev?.courses.find((c) => c.code === code)?.grade;
  const regraded = !!prev && (prev.regraded || parsed.courses.some((c) => grade(c.code) !== undefined && grade(c.code) !== c.grade));
  const next = { ...parsed, importedAt: now.toISOString(), ...(regraded && { regraded: true as const }) };
  return { ...state, results: [...state.results.filter((r) => r.label !== parsed.label), next] };
}

export function finishExams(state: AppState, label: string): AppState {
  return state.examsDone.includes(label) ? state : { ...state, examsDone: [...state.examsDone, label] };
}

export function setGraduated(state: AppState, now: Date): AppState {
  return { ...state, graduated: now.toISOString() };
}
```

```ts
import { useState } from "preact/hooks";
import { importResult } from "../domain/actions";
import type { ParsedResult } from "../iium/results";
import { readResultSlip } from "../iium/sync";
import { dispatch } from "../store/store";

export function useResultImport() {
  const [busy, setBusy] = useState(false);
  const [error, setError] = useState<Error | null>(null);
  const [preview, setPreview] = useState<ParsedResult | null>(null);

  async function start(file: Blob) {
    setBusy(true);
    setError(null);
    try {
      setPreview(await readResultSlip(file));
    } catch (e) {
      setError(e as Error);
    } finally {
      setBusy(false);
    }
  }

  function confirm() {
    if (!preview) return;
    const parsed = preview;
    dispatch((s) => importResult(s, parsed, new Date()));
    setPreview(null);
  }

  return { busy, error, preview, start, confirm, cancel: () => setPreview(null) };
}
```

```ts
import { eloFromGpa, gpa, labelKey, sortResults } from "./grades";
import { ymd } from "./time";
import type { AppState, Task } from "./types";

export const DELTA = { onTime: 15, finalOnTime: 45, late: -10, missed: -20, finalMissed: -60 };
const UNRATED = 900;

/** Tasks ticked before completion times were recorded count as on time. */
export const onTime = (t: Task) => t.done && (!t.completedAt || t.completedAt <= t.due);

export interface EloChange {
  taskId: string;
  title: string;
  delta: number;
  at: string; // ISO, when the change happened
}

export interface Season {
  rated: boolean;
  placement: number;
  elo: number;
  changes: EloChange[]; // newest first
}

/** This semester's Elo: placement from the last earlier result slip, moved by tasks due this semester. */
export function seasonElo(state: AppState, now: Date): Season {
  const sem = state.semester;
  const prior = sortResults(state.results).filter((r) => !sem || labelKey(r.label) < labelKey(sem.label));
  const last = prior.at(-1);
  const placement = last ? eloFromGpa(gpa(last.courses).gpa) : UNRATED;
  const changes: EloChange[] = [];
  for (const t of state.tasks) {
    if (sem && ymd(new Date(t.due)) < sem.start) continue;
    let delta = 0;
    if (t.done) delta = onTime(t) ? (t.final ? DELTA.finalOnTime : DELTA.onTime) : DELTA.late;
    else if (new Date(t.due) < now) delta = t.final ? DELTA.finalMissed : DELTA.missed;
    if (delta) changes.push({ taskId: t.id, title: t.title, delta, at: (t.done && t.completedAt) || t.due });
  }
  changes.sort((a, b) => b.at.localeCompare(a.at));
  const elo = Math.max(0, changes.reduce((n, c) => n + c.delta, placement));
  return { rated: !!last, placement, elo, changes };
}
```

```tsx
      return;
    }
    const fresh = ids.filter((id) => !seen.current!.has(id));
    fresh.forEach((id) => seen.current!.add(id));
    if (fresh.length) setQueue((q) => [...q, ...fresh]);
  }, [achievements]);

  useEffect(() => {
    if (!queue.length) return;
    const t = setTimeout(() => setQueue((q) => q.slice(1)), 4000);
    return () => clearTimeout(t);
  }, [queue]);

  const a = ACHIEVEMENTS.find((x) => x.id === queue[0]);
  if (!a) return null;
  return (
    <div class={`adv-toast ${a.secret ? "secret" : ""}`} role="status">
      <b class="pixel">{a.secret ? "Challenge Complete!" : "Advancement Made!"}</b>
      <span class="pixel">{a.name}</span>
    </div>
  );
}
```
